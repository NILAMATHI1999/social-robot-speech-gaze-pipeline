#   ~/vlm_venv/bin/pip install --dry-run transformers accelerate pillow num2words   # check: no nvidia-* / cuda-* / torch-*

#   ~/vlm_venv/bin/python -c "import torch, transformers; print(torch.__version__, transformers.__version__)"
